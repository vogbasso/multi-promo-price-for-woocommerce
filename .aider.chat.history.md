/**
 * Check if WooCommerce is active.
 *
 * @return bool
 */
function mppw_is_woocommerce_active() {
	return in_array( 'woocommerce/woocommerce.php', apply_filters( 'active_plugins', get_option( 'active_plugins' ) ), true )
		|| ( is_multisite() && array_key_exists( 'woocommerce/woocommerce.php', get_site_option( 'active_sitewide_plugins', array() ) ) );

/**
 * Initialize the plugin.

function mppw_init() {
	if ( ! mppw_is_woocommerce_active() ) {
		add_action( 'admin_notices', 'mppw_missing_wc_notice' );
		return;
	}

	// Admin product fields (Simple products).

	add_action( 'admin_notices', 'mppw_display_admin_notices' );

	add_filter( 'woocommerce_product_get_price', 'mppw_filter_product_price', 20, 2 );

	mppw_process_and_save_meta( $post_id, $price_raw, $from_raw, $to_raw, $product->get_name() );

/**
 * Helper to validate and save flash promo fields.
 *
 * @param int    $object_id
 * @param string $price_raw
 * @param string $from_raw
 * @param string $to_raw
 * @param string $label
 */
function mppw_process_and_save_meta( $object_id, $price_raw, $from_raw, $to_raw, $label = '' ) {
	$clean_price = ( '' !== $price_raw ) ? wc_format_decimal( $price_raw ) : '';

	if ( '' !== $clean_price ) {
		// Strict validation: End date is required if flash sale price is entered.
		if ( empty( $to_raw ) ) {
			mppw_add_admin_notice(
				sprintf(
					/* translators: %s: product title */
					__( 'Erreur pour "%s" : La date de fin est obligatoire si le prix super promo est renseigné. Le prix super promo n\'a pas été enregistré.', 'multi-promo-price-for-woocommerce' ),
					$label ? $label : '#' . $object_id
				),
				'error'
			);

			// Delete super promo price to ensure strict consistency.
			delete_post_meta( $object_id, '_mppw_sale_price' );
			delete_post_meta( $object_id, '_mppw_sale_price_dates_from' );
			delete_post_meta( $object_id, '_mppw_sale_price_dates_to' );
			return;
		}

		$timestamp_to = strtotime( $to_raw . ' 23:59:59' );
		update_post_meta( $object_id, '_mppw_sale_price', $clean_price );
		update_post_meta( $object_id, '_mppw_sale_price_dates_to', $timestamp_to );

		if ( ! empty( $from_raw ) ) {
			$timestamp_from = strtotime( $from_raw . ' 00:00:00' );
			update_post_meta( $object_id, '_mppw_sale_price_dates_from', $timestamp_from );
		} else {
			delete_post_meta( $object_id, '_mppw_sale_price_dates_from' );
		}
	} else {
		delete_post_meta( $object_id, '_mppw_sale_price' );
		delete_post_meta( $object_id, '_mppw_sale_price_dates_from' );
		delete_post_meta( $object_id, '_mppw_sale_price_dates_to' );
	}
}

/**
 * Queue an admin notice in a transient.
 *
 * @param string $message
 * @param string $type
 */
function mppw_add_admin_notice( $message, $type = 'error' ) {
	$user_id = get_current_user_id();
	$notices = get_transient( 'mppw_admin_notices_' . $user_id );
	if ( ! is_array( $notices ) ) {
		$notices = array();

		'message' => $message,
		'type'    => $type,
	);
	set_transient( 'mppw_admin_notices_' . $user_id, $notices, 60 );
}

/**
 * Check if the flash super promo is currently active for a given product or variation.
 *
 * @param int $product_id
 * @return bool
 */
function mppw_is_super_promo_active( $product_id ) {
	$sale_price = get_post_meta( $product_id, '_mppw_sale_price', true );
	if ( '' === $sale_price || null === $sale_price || false === $sale_price ) {
		return false;
	}

	$date_to = get_post_meta( $product_id, '_mppw_sale_price_dates_to', true );
	if ( empty( $date_to ) ) {
		return false;
	}

	$now       = current_time( 'timestamp' );
	$date_from = get_post_meta( $product_id, '_mppw_sale_price_dates_from', true );

	// If start date is set, check if promo has started.
	if ( ! empty( $date_from ) && $now < (int) $date_from ) {
		return false;
	}

	// Check if promo has expired.
	if ( $now > (int) $date_to ) {
		return false;
	}

	return true;
}

/**
 * Get active super promo price for a product.
 *
 * @param int $product_id
 * @return string|false
 */
function mppw_get_active_super_promo_price( $product_id ) {
	if ( mppw_is_super_promo_active( $product_id ) ) {
		return get_post_meta( $product_id, '_mppw_sale_price', true );
	}
	return false;
}

/**
 * Filter product/variation price.
 *
 * @param string     $price
 * @param WC_Product $product
 * @return string
 */
function mppw_filter_product_price( $price, $product ) {
	if ( ! $product instanceof WC_Product ) {
		return $price;
	}

	$super_price = mppw_get_active_super_promo_price( $product->get_id() );
	if ( false !== $super_price && '' !== $super_price ) {
		return (string) $super_price;
	}

	return $price;
}

/**
 * Filter product/variation sale price.
 *
 * @param string     $sale_price
 * @param WC_Product $product
 * @return string
 */
function mppw_filter_product_sale_price( $sale_price, $product ) {
	if ( ! $product instanceof WC_Product ) {
		return $sale_price;
	}

	$super_price = mppw_get_active_super_promo_price( $product->get_id() );
	if ( false !== $super_price && '' !== $super_price ) {
		return (string) $super_price;
	}

	return $sale_price;
}

/**
 * Filter is_on_sale to return true if super promo is currently active.
 *
 * @param bool       $is_on_sale
 * @param WC_Product $product
 * @return bool
 */
function mppw_filter_is_on_sale( $is_on_sale, $product ) {
	if ( ! $product instanceof WC_Product ) {
		return $is_on_sale;
	}

	if ( mppw_is_super_promo_active( $product->get_id() ) ) {
		return true;
	}

	return $is_on_sale;
}

/**
 * Process validation and save metadata for product or variation.
 *
 * @param int    $object_id
 * @param string $price_raw
 * @param string $from_raw
 * @param string $to_raw
 * @param string $label
 */
function mppw_process_and_save_meta( $object_id, $price_raw, $from_raw, $to_raw, $label = '' ) {
	$clean_price = ( '' !== $price_raw ) ? wc_format_decimal( $price_raw ) : '';

	if ( '' !== $clean_price ) {
		// Strict validation: End date is required if secondary sale price is set.
		if ( empty( $to_raw ) ) {
			mppw_add_admin_notice(
				sprintf(
					/* translators: %s: product title */
					__( 'Erreur pour "%s" : La date de fin est obligatoire si le prix promo secondaire est renseigné. Le prix n\'a pas été enregistré.', 'multi-promo-price-for-woocommerce' ),
					$label ? $label : '#' . $object_id
				),
				'error'
			);

			// Do not save secondary promo price if end date is missing.
			delete_post_meta( $object_id, '_mppw_sale_price' );
			delete_post_meta( $object_id, '_mppw_dates_from' );
			delete_post_meta( $object_id, '_mppw_dates_to' );
			return;
		}

		$timestamp_to = strtotime( $to_raw . ' 23:59:59' );
		update_post_meta( $object_id, '_mppw_sale_price', $clean_price );
		update_post_meta( $object_id, '_mppw_dates_to', $timestamp_to );

		if ( ! empty( $from_raw ) ) {
			$timestamp_from = strtotime( $from_raw . ' 00:00:00' );
			update_post_meta( $object_id, '_mppw_dates_from', $timestamp_from );
		} else {
			delete_post_meta( $object_id, '_mppw_dates_from' );
		}
	} else {
		delete_post_meta( $object_id, '_mppw_sale_price' );
		delete_post_meta( $object_id, '_mppw_dates_from' );
		delete_post_meta( $object_id, '_mppw_dates_to' );
	}
}

/**
 * Check if the secondary promo is currently active for a given product or variation.
 *
 * @param int $product_id
 * @return bool
 */
function mppw_is_super_promo_active( $product_id ) {
	$sale_price = get_post_meta( $product_id, '_mppw_sale_price', true );
	if ( '' === $sale_price || null === $sale_price || false === $sale_price ) {
		return false;
	}

	$date_to = get_post_meta( $product_id, '_mppw_dates_to', true );
	if ( empty( $date_to ) ) {
		return false;
	}

	$now       = current_time( 'timestamp' );
	$date_from = get_post_meta( $product_id, '_mppw_dates_from', true );

	// Promo must be started if start date is defined (00:00:00).
	if ( ! empty( $date_from ) && $now < (int) $date_from ) {
		return false;
	}

	// Promo must not have expired (23:59:59).
	if ( $now > (int) $date_to ) {
		return false;
	}

	return true;
}

(function () {
	'use strict';

	/**
	 * Pads single digits with leading zero.
	 */
	function padZero(num) {
		return num < 10 ? '0' + num : num;
	}

	/**
	 * Initializes timer element.
	 */
	function initTimer(wrapper) {
		if (!wrapper) {
			return;
		}

		var daysEl = wrapper.querySelector('[data-unit="days"]');
		var hoursEl = wrapper.querySelector('[data-unit="hours"]');
		var minsEl = wrapper.querySelector('[data-unit="mins"]');
		var secsEl = wrapper.querySelector('[data-unit="secs"]');

		var intervalId = null;

		function update() {
			var expireTimestamp = parseInt(wrapper.getAttribute('data-expire'), 10);
			if (!expireTimestamp || isNaN(expireTimestamp)) {
				wrapper.style.display = 'none';
				if (intervalId) {
					clearInterval(intervalId);
				}
				return;
			}

			// Current Unix timestamp in seconds
			var now = Math.floor(Date.now() / 1000);
			var diff = expireTimestamp - now;

			if (diff <= 0) {
				wrapper.style.display = 'none';
				if (intervalId) {
					clearInterval(intervalId);
				}
				// Reload the page smoothly when promotion expires
				setTimeout(function () {
					window.location.reload();
				}, 1000);
				return;
			}

			wrapper.style.display = '';

			var days = Math.floor(diff / 86400);
			var hours = Math.floor((diff % 86400) / 3600);
			var mins = Math.floor((diff % 3600) / 60);
			var secs = Math.floor(diff % 60);

			if (daysEl) daysEl.textContent = padZero(days);
			if (hoursEl) hoursEl.textContent = padZero(hours);
			if (minsEl) minsEl.textContent = padZero(mins);
			if (secsEl) secsEl.textContent = padZero(secs);
		}

		update();
		intervalId = setInterval(update, 1000);
		wrapper._mppwIntervalId = intervalId;
	}

	/**
	 * Handles variable products changes.
	 */
	function bindVariationEvents(wrapper) {
		if (!wrapper) {
			return;
		}

		var form = document.querySelector('form.variations_form');
		if (!form || typeof jQuery === 'undefined') {
			return;
		}

		jQuery(form).on('found_variation', function (event, variation) {
			if (variation && variation.mppw_expire_timestamp) {
				wrapper.setAttribute('data-expire', variation.mppw_expire_timestamp);
				initTimer(wrapper);
			} else {
				wrapper.setAttribute('data-expire', '');
				wrapper.style.display = 'none';
			}
		});

		jQuery(form).on('reset_data', function () {
			var defaultExpire = wrapper.getAttribute('data-default-expire') || '';
			wrapper.setAttribute('data-expire', defaultExpire);
			if (defaultExpire) {
				initTimer(wrapper);
			} else {
				wrapper.style.display = 'none';
			}
		});
	}

	document.addEventListener('DOMContentLoaded', function () {
		var wrappers = document.querySelectorAll('.mppw-countdown-wrapper');
		wrappers.forEach(function (wrapper) {
			initTimer(wrapper);
			bindVariationEvents(wrapper);
		});
	});
})();

/**
 * Process validation and save metadata for product or variation.
 *
 * @param int    $object_id
 * @param string $price_raw
 * @param string $from_raw
 * @param string $to_raw
 * @param string $label
 */
function mppw_process_and_save_meta( $object_id, $price_raw, $from_raw, $to_raw, $label = '' ) {
	$clean_price = ( '' !== $price_raw ) ? wc_format_decimal( $price_raw ) : '';

	if ( '' !== $clean_price ) {
		// Strict validation: End date is required if secondary sale price is set.
		if ( empty( $to_raw ) ) {
			mppw_add_admin_notice(
				sprintf(
					/* translators: %s: product title */
					__( 'Erreur pour "%s" : La date de fin est obligatoire si le prix promo secondaire est renseigné. Le prix n\'a pas été enregistré.', 'multi-promo-price-for-woocommerce' ),
					$label ? $label : '#' . $object_id
				),
				'error'
			);

			// Do not save secondary promo price if end date is missing.
			delete_post_meta( $object_id, '_mppw_sale_price' );
			delete_post_meta( $object_id, '_mppw_dates_from' );
			delete_post_meta( $object_id, '_mppw_dates_to' );
			return;
		}

		$timestamp_to = strtotime( $to_raw . ' 23:59:59' );
		update_post_meta( $object_id, '_mppw_sale_price', $clean_price );
		update_post_meta( $object_id, '_mppw_dates_to', $timestamp_to );

		if ( ! empty( $from_raw ) ) {
			$timestamp_from = strtotime( $from_raw . ' 00:00:00' );
			update_post_meta( $object_id, '_mppw_dates_from', $timestamp_from );
		} else {
			delete_post_meta( $object_id, '_mppw_dates_from' );
		}
	} else {
		delete_post_meta( $object_id, '_mppw_sale_price' );
		delete_post_meta( $object_id, '_mppw_dates_from' );
		delete_post_meta( $object_id, '_mppw_dates_to' );
	}
}

/**
 * Check if the secondary promo is currently active for a given product or variation.
 *
 * @param int $product_id
 * @return bool
 */
function mppw_is_super_promo_active( $product_id ) {
	$sale_price = get_post_meta( $product_id, '_mppw_sale_price', true );
	if ( '' === $sale_price || null === $sale_price || false === $sale_price ) {
		return false;
	}

	$date_to = get_post_meta( $product_id, '_mppw_dates_to', true );
	if ( empty( $date_to ) ) {
		return false;
	}

	$now       = current_time( 'timestamp' );
	$date_from = get_post_meta( $product_id, '_mppw_dates_from', true );

	// Promo must be started if start date is defined (00:00:00).
	if ( ! empty( $date_from ) && $now < (int) $date_from ) {
		return false;
	}

	// Promo must not have expired (23:59:59).
	if ( $now > (int) $date_to ) {
		return false;
	}

	return true;
}

(function () {
	'use strict';

	/**
	 * Pads single digits with leading zero.
	 */
	function padZero(num) {
		return num < 10 ? '0' + num : num;
	}

	/**
	 * Updates the timer header label.
	 */
	function updateTimerTitle(wrapper, title) {
		if (!wrapper) {
			return;
		}
		var titleEl = wrapper.querySelector('.mppw-countdown-title');
		if (titleEl && title) {
			titleEl.textContent = title;
		}
	}

	/**
	 * Initializes timer element.
	 */
	function initTimer(wrapper) {
		if (!wrapper) {
			return;
		}

		var daysEl = wrapper.querySelector('[data-unit="days"]');
		var hoursEl = wrapper.querySelector('[data-unit="hours"]');
		var minsEl = wrapper.querySelector('[data-unit="mins"]');
		var secsEl = wrapper.querySelector('[data-unit="secs"]');

		if (wrapper._mppwIntervalId) {
			clearInterval(wrapper._mppwIntervalId);
			wrapper._mppwIntervalId = null;
		}

		function update() {
			var expireTimestamp = parseInt(wrapper.getAttribute('data-expire'), 10);
			if (!expireTimestamp || isNaN(expireTimestamp)) {
				wrapper.style.display = 'none';
				if (wrapper._mppwIntervalId) {
					clearInterval(wrapper._mppwIntervalId);
					wrapper._mppwIntervalId = null;
				}
				return;
			}

			// Current Unix timestamp in seconds
			var now = Math.floor(Date.now() / 1000);
			var diff = expireTimestamp - now;

			if (diff <= 0) {
				wrapper.style.display = 'none';
				if (wrapper._mppwIntervalId) {
					clearInterval(wrapper._mppwIntervalId);
					wrapper._mppwIntervalId = null;
				}
				// Reload page when promotion reaches zero
				setTimeout(function () {
					window.location.reload();
				}, 1000);
				return;
			}

			wrapper.style.display = '';

			var days = Math.floor(diff / 86400);
			var hours = Math.floor((diff % 86400) / 3600);
			var mins = Math.floor((diff % 3600) / 60);
			var secs = Math.floor(diff % 60);

			if (daysEl) daysEl.textContent = padZero(days);
			if (hoursEl) hoursEl.textContent = padZero(hours);
			if (minsEl) minsEl.textContent = padZero(mins);
			if (secsEl) secsEl.textContent = padZero(secs);
		}

		update();
		wrapper._mppwIntervalId = setInterval(update, 1000);
	}

	/**
	 * Handles variable products changes.
	 */
	function bindVariationEvents(wrapper) {
		if (!wrapper) {
			return;
		}

		var form = document.querySelector('form.variations_form');
		if (!form || typeof jQuery === 'undefined') {
			return;
		}

		jQuery(form).on('found_variation', function (event, variation) {
			if (variation && variation.mppw_expire_timestamp) {
				wrapper.setAttribute('data-expire', variation.mppw_expire_timestamp);
				if (variation.mppw_countdown_title) {
					updateTimerTitle(wrapper, variation.mppw_countdown_title);
				}
				initTimer(wrapper);
			} else {
				wrapper.setAttribute('data-expire', '');
				wrapper.style.display = 'none';
				if (wrapper._mppwIntervalId) {
					clearInterval(wrapper._mppwIntervalId);
					wrapper._mppwIntervalId = null;
				}
			}
		});

		jQuery(form).on('reset_data', function () {
			var defaultExpire = wrapper.getAttribute('data-default-expire') || '';
			var defaultTitle  = wrapper.getAttribute('data-default-title') || '';

			wrapper.setAttribute('data-expire', defaultExpire);
			if (defaultTitle) {
				updateTimerTitle(wrapper, defaultTitle);
			}

			if (defaultExpire) {
				initTimer(wrapper);
			} else {
				wrapper.style.display = 'none';
				if (wrapper._mppwIntervalId) {
					clearInterval(wrapper._mppwIntervalId);
					wrapper._mppwIntervalId = null;
				}
			}
		});
	}

	document.addEventListener('DOMContentLoaded', function () {
		var wrappers = document.querySelectorAll('.mppw-countdown-wrapper');
		wrappers.forEach(function (wrapper) {
			initTimer(wrapper);
			bindVariationEvents(wrapper);
		});
	});
})();

/**
 * Process validation and save metadata for product or variation.
 *
 * @param int    $object_id
 * @param string $price_raw
 * @param string $from_raw
 * @param string $to_raw
 * @param string $label
 */
function mppw_process_and_save_meta( $object_id, $price_raw, $from_raw, $to_raw, $label = '' ) {
	$clean_price = ( '' !== $price_raw ) ? wc_format_decimal( $price_raw ) : '';

	if ( '' !== $clean_price ) {
		// Strict validation: End date is required if secondary sale price is set.
		if ( empty( $to_raw ) ) {
			mppw_add_admin_notice(
				sprintf(
					/* translators: %s: product title */
					__( 'Erreur pour "%s" : La date de fin est obligatoire si le prix promo secondaire est renseigné. Le prix n\'a pas été enregistré.', 'multi-promo-price-for-woocommerce' ),
					$label ? $label : '#' . $object_id
				),
				'error'
			);

			// Do not save secondary promo price if end date is missing.
			delete_post_meta( $object_id, '_mppw_sale_price' );
			delete_post_meta( $object_id, '_mppw_dates_from' );
			delete_post_meta( $object_id, '_mppw_dates_to' );
			return;
		}

		$timestamp_to = strtotime( $to_raw . ' 23:59:59' );
		update_post_meta( $object_id, '_mppw_sale_price', $clean_price );
		update_post_meta( $object_id, '_mppw_dates_to', $timestamp_to );

		if ( ! empty( $from_raw ) ) {
			$timestamp_from = strtotime( $from_raw . ' 00:00:00' );
			update_post_meta( $object_id, '_mppw_dates_from', $timestamp_from );
		} else {
			delete_post_meta( $object_id, '_mppw_dates_from' );
		}
	} else {
		delete_post_meta( $object_id, '_mppw_sale_price' );
		delete_post_meta( $object_id, '_mppw_dates_from' );
		delete_post_meta( $object_id, '_mppw_dates_to' );
	}
}

/**
 * Check if the secondary promo is currently active for a given product or variation.
 *
 * @param int $product_id
 * @return bool
 */
function mppw_is_super_promo_active( $product_id ) {
	$sale_price = get_post_meta( $product_id, '_mppw_sale_price', true );
	if ( '' === $sale_price || null === $sale_price || false === $sale_price ) {
		return false;
	}

	$date_to = get_post_meta( $product_id, '_mppw_dates_to', true );
	if ( empty( $date_to ) ) {
		return false;
	}

	$now       = current_time( 'timestamp' );
	$date_from = get_post_meta( $product_id, '_mppw_dates_from', true );

	// Promo must be started if start date is defined (00:00:00).
	if ( ! empty( $date_from ) && $now < (int) $date_from ) {
		return false;
	}

	// Promo must not have expired (23:59:59).
	if ( $now > (int) $date_to ) {
		return false;
	}

	return true;
}

/**
 * Retrieve end timestamp for standard WooCommerce sale if active.
 * Ensures the expiration ends at 23:59:59 of the day if no specific time is provided.
 *
 * @param WC_Product $product
 * @return int|false
 */
function mppw_get_standard_promo_end_timestamp( $product ) {
	if ( ! $product instanceof WC_Product ) {
		return false;
	}

	$sale_price = $product->get_sale_price( 'edit' );
	if ( '' === $sale_price || null === $sale_price || false === $sale_price ) {
		return false;
	}

	$date_to = $product->get_date_on_sale_to( 'edit' );
	$timestamp_to = 0;

	if ( $date_to instanceof WC_DateTime ) {
		$timestamp_to = $date_to->getTimestamp();
	} else {
		$raw_to = get_post_meta( $product->get_id(), '_sale_price_dates_to', true );
		if ( ! empty( $raw_to ) ) {
			$timestamp_to = is_numeric( $raw_to ) ? (int) $raw_to : strtotime( $raw_to );
		}
	}

	if ( empty( $timestamp_to ) ) {
		return false;
	}

	// If the timestamp has no time component (midnight 00:00:00), adjust it to 23:59:59 of that day.
	if ( 0 === ( $timestamp_to % 86400 ) ) {
		$timestamp_to += 86399;
	}

	$now = current_time( 'timestamp' );

	// Check start date if present
	$date_from = $product->get_date_on_sale_from( 'edit' );
	$timestamp_from = 0;
	if ( $date_from instanceof WC_DateTime ) {
		$timestamp_from = $date_from->getTimestamp();
	} else {
		$raw_from = get_post_meta( $product->get_id(), '_sale_price_dates_from', true );
		if ( ! empty( $raw_from ) ) {
			$timestamp_from = is_numeric( $raw_from ) ? (int) $raw_from : strtotime( $raw_from );
		}
	}

	if ( ! empty( $timestamp_from ) && $now < $timestamp_from ) {
		return false;
	}

	if ( $now > $timestamp_to ) {
		return false;
	}

	return $timestamp_to;
}

/**
 * Determine countdown information in cascading order:
 * 1. Super Promo Flash
 * 2. Standard WooCommerce Promo with end date
 * 3. None
 *
 * @param WC_Product|int $product_or_id
 * @return array|false Returns array with 'expire_timestamp' and 'title', or false.
 */
function mppw_get_active_countdown_info( $product_or_id ) {
	$product = is_numeric( $product_or_id ) ? wc_get_product( $product_or_id ) : $product_or_id;
	if ( ! $product instanceof WC_Product ) {
		return false;
	}

	$product_id = $product->get_id();

	// Case 1: Super Promo Flash active.
	if ( mppw_is_super_promo_active( $product_id ) ) {
		$flash_to = (int) get_post_meta( $product_id, '_mppw_dates_to', true );
		if ( $flash_to > current_time( 'timestamp' ) ) {
			return array(
				'type'             => 'flash',
				'expire_timestamp' => $flash_to,
				'title'            => __( 'Vente Flash se termine dans :', 'multi-promo-price-for-woocommerce' ),
			);
		}
	}

	// Case 2: Standard WooCommerce Promo active with scheduled end date.
	$standard_to = mppw_get_standard_promo_end_timestamp( $product );
	if ( ! empty( $standard_to ) && $standard_to > current_time( 'timestamp' ) ) {
		return array(
			'type'             => 'standard',
			'expire_timestamp' => $standard_to,
			'title'            => __( 'Offre promotionnelle se termine dans :', 'multi-promo-price-for-woocommerce' ),
		);
	}

	// Case 3: No promotion with end date.
	return false;
}

/**
 * Check if the product or any of its variations has an active countdown (flash or standard).
 *
 * @param WC_Product $product
 * @return bool
 */
function mppw_product_or_children_has_active_promo( $product ) {
	if ( ! $product instanceof WC_Product ) {
		return false;
	}

	if ( false !== mppw_get_active_countdown_info( $product ) ) {
		return true;
	}

	if ( $product->is_type( 'variable' ) ) {
		$children = $product->get_children();
		if ( ! empty( $children ) ) {
			foreach ( $children as $child_id ) {
				if ( false !== mppw_get_active_countdown_info( $child_id ) ) {
					return true;
				}
			}
		}
	}

	return false;
}

# Multi Promo Price for WooCommerce

Extension WooCommerce permettant d'ajouter une promotion secondaire (promo flash) avec date d'expiration obligatoire par-dessus une promotion standard sans date de fin.

## Fonctionnalités
- Promotion flash secondaire prioritaire sur la promotion standard WooCommerce.
- Date de fin obligatoire avec expiration automatique à 23:59:59 pour les promos flash.
- **Compte à rebours en cascade (Sale Countdown Timer)** :
  - **Cas 1 (Vente Flash)** : Si la super promo flash est active avec sa date de fin, affichage du compte à rebours ciblant cette date avec le titre « Vente Flash se termine dans : ».
  - **Cas 2 (Promo standard WooCommerce)** : Si aucune vente flash n'est active mais que la promo standard WooCommerce dispose d'une date de fin (`_sale_price_dates_to`), affichage automatique du compte à rebours avec le titre « Offre promotionnelle se termine dans : » (calée à 23:59:59 si l'heure n'est pas spécifiée).
  - **Cas 3** : Masquage automatique du compte à rebours si aucune date de fin n'est programmée.
- Compatible avec les produits simples et les variations de produits (mise à jour dynamique en Vanilla JS lors du changement de variation).
- Chargement optimisé des styles et scripts uniquement sur les pages produits avec promo active.
